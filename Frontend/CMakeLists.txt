cmake_minimum_required(VERSION 3.16)
project(EmailClusteringFrontend)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
set(CMAKE_AUTOUIC ON)

find_package(Qt6 COMPONENTS
    Core
    Gui
    Widgets
    Charts
    Concurrent
    REQUIRED
)

include_directories(${CMAKE_CURRENT_SOURCE_DIR}/../Backend/src)
include_directories(${CMAKE_CURRENT_SOURCE_DIR}/../Backend/include)

set(PROJECT_SOURCES
    src/main.cpp
    src/mainwindow.cpp
    src/mainwindow.h
    src/datavisualization.cpp
    src/datavisualization.h
    src/filemanager.cpp
    src/filemanager.h
    src/resources.qrc
)

# On macOS, produce a proper .app bundle (rather than a bare executable) so
# it can carry an icon, an Info.plist, and be deployed with macdeployqt.
# ASSETS_ICNS is generated by deploy.sh from assets/icon.png before this
# CMakeLists.txt runs; if it's not there yet (e.g. a plain dev build), the
# bundle is still created, just without a custom icon.
set(ASSETS_DIR ${CMAKE_CURRENT_SOURCE_DIR}/assets)
set(ASSETS_ICNS ${ASSETS_DIR}/AppIcon.icns)

if(APPLE AND EXISTS ${ASSETS_ICNS})
    set(PROJECT_SOURCES ${PROJECT_SOURCES} ${ASSETS_ICNS})
    set_source_files_properties(${ASSETS_ICNS} PROPERTIES
        MACOSX_PACKAGE_LOCATION "Resources"
    )
endif()

add_executable(${PROJECT_NAME} MACOSX_BUNDLE ${PROJECT_SOURCES})

if(APPLE)
    set_target_properties(${PROJECT_NAME} PROPERTIES
        MACOSX_BUNDLE_BUNDLE_NAME "Email Clustering Analysis Tool"
        MACOSX_BUNDLE_GUI_IDENTIFIER "com.davidyatsevich.emailclustering"
        MACOSX_BUNDLE_SHORT_VERSION_STRING "1.0.0"
        MACOSX_BUNDLE_BUNDLE_VERSION "1.0.0"
    )
    if(EXISTS ${ASSETS_ICNS})
        set_target_properties(${PROJECT_NAME} PROPERTIES
            MACOSX_BUNDLE_ICON_FILE "AppIcon.icns"
        )
    endif()
endif()

target_link_libraries(${PROJECT_NAME}
    Qt6::Core
    Qt6::Gui
    Qt6::Widgets
    Qt6::Charts
    Qt6::Concurrent
    pipeline
)
